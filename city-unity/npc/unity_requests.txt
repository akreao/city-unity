// The board. divine-pride lists 54,35, which is inside the wall in the map
// data, so it stands at 53,35 in line with the manager and clerk. The kRO
// client's navigation table doesn't list it, HF's /navi gives 60,35.
// [unverified: board position and sprite]
moc_para01,53,35,4	script	City Merchant Union Board#cu	857,{
	.@npc$ = "Unity Requests#cu";
	mes "[City Merchant Union Board]";
	mes "Requests posted by the City Merchant Union.";
	if (isbegin_quest(12370) != 1) {
		mes "Only union members may take them. The union's manager stands nearby.";
		close;
	}
	next;
	switch (select("Wanted bounty (weekly):A-rank delivery:B-rank delivery:C-rank delivery:Close")) {
	case 1:
		mes "[City Merchant Union Board]";
		.@n = getarraysize(getvariableofnpc(.w_quest, .@npc$));
		for (.@i = 0; .@i < .@n; .@i++) {
			.@q = getvariableofnpc(.w_quest[.@i], .@npc$);
			if (isbegin_quest(.@q) == 1) {
				mes "You already hold a bounty. Check your quest log, and report to the clerk once it's done.";
				close;
			}
		}
		if (isbegin_quest(5470) == 1) {
			if (gettimetick(2) < unity_wanted_reset) {
				mes "There's no new bounty yet. New ones are posted on Wednesday at 4 a.m.";
				close;
			}
			erasequest 5470;
		}
		.@i = rand(.@n);
		.@q = getvariableofnpc(.w_quest[.@i], .@npc$);
		setquest .@q;
		mes "You take this week's bounty: defeat ^0000FF" + getmonsterinfo(getvariableofnpc(.w_boss[.@i], .@npc$), MOB_NAME) + "^000000 for " + getvariableofnpc(.w_reward[.@i], .@npc$) + " Union Insignias.";
		mes "Report to the clerk when it's done.";
		close;
	case 2:
	case 3:
	case 4:
		.@r = @menu - 2;
		.@rk$ = getvariableofnpc(.rank$[.@r], .@npc$);
		mes "[City Merchant Union Board]";
		if (callfunc("F_Unity_HeldDelivery") > 0) {
			mes "You already hold a delivery. Finish it, or give it back to the clerk, before taking another.";
			close;
		}
		.@sb = getvariableofnpc(.standby[.@r], .@npc$);
		if (isbegin_quest(.@sb) == 1) {
			if (checkquest(.@sb, PLAYTIME) == 2)
				erasequest .@sb;
			else {
				mes "You've done today's " + .@rk$ + "-rank delivery. New requests are posted at 4 a.m.";
				close;
			}
		}
		.@n = getarraysize(getvariableofnpc(getd(".d" + .@rk$ + "_quest"), .@npc$));
		.@i = rand(.@n);
		.@q = getvariableofnpc(getd(".d" + .@rk$ + "_quest[" + .@i + "]"), .@npc$);
		.@item = getvariableofnpc(getd(".d" + .@rk$ + "_item[" + .@i + "]"), .@npc$);
		.@amt = getvariableofnpc(getd(".d" + .@rk$ + "_amount[" + .@i + "]"), .@npc$);

		mes "You take a " + .@rk$ + "-rank delivery: " + .@amt + " " + getitemname(.@item) + ".";
		mes "Your quest log says who it's for and where they are. Bring the receipt back to the clerk.";
		close;
	case 5:
		close;
	}
	end;

OnInit:
	if (callfunc("F_Unity_StandDown"))
		disablenpc strnpcinfo(0);
	end;
}

// Returns the delivery task or receipt quest the player holds, or 0.
function	script	F_Unity_HeldDelivery	{
	for (.@q = 5497; .@q <= 5726; .@q++) {
		if (.@q == 5577 || .@q == 5676)
			continue;
		if (isbegin_quest(.@q) == 1)
			return .@q;
	}
	return 0;
}

// Clerk Gabyunus (kRO 사무관 가뷰너스, iRO "Secretary Governor"):
// sprite 893 at 53,32 in both clients' navigation tables.
moc_para01,53,32,4	script	Secretary Governor#cu	893,{
	.@npc$ = "Unity Requests#cu";
	.@n$ = "[Secretary Governor]";
	mes .@n$;
	mes "I settle the City Merchant Union's requests.";
	next;
	switch (select("Report a wanted bounty:Report a delivery:Give back a request:Nothing")) {
	case 1:
		.@n = getarraysize(getvariableofnpc(.w_quest, .@npc$));
		for (.@i = 0; .@i < .@n; .@i++) {
			.@q = getvariableofnpc(.w_quest[.@i], .@npc$);
			if (isbegin_quest(.@q) == 1)
				break;
		}
		mes .@n$;
		if (.@i == .@n) {
			mes "You don't hold a bounty.";
			close;
		}
		if (checkquest(.@q, HUNTING) != 2) {
			mes "The target isn't down yet. Come back once it is.";
			close;
		}
		.@reward = getvariableofnpc(.w_reward[.@i], .@npc$);
		if (checkweight(25051, .@reward) == 0) {
			mes "You're carrying too much to take the reward.";
			close;
		}
		erasequest .@q;
		callfunc "F_Unity_WantedWait";
		getitem 25051, .@reward;
		mes "Splendid work. Here are your " + .@reward + " Union Insignias.";
		mes "The next bounty is posted on Wednesday at 4 a.m.";
		close;
	case 2:
		.@q = callfunc("F_Unity_HeldDelivery");
		mes .@n$;
		if (.@q == 0) {
			mes "You don't hold a delivery.";
			close;
		}
		.@r = (.@q < 5577) ? 0 : ((.@q < 5676) ? 1 : 2);
		.@rk$ = getvariableofnpc(.rank$[.@r], .@npc$);
		.@task = callfunc("F_Unity_DeliveryTask", .@q);
		if (.@task == .@q) {
			mes "Deliver the goods first, then bring me the receipt.";
			close;
		}
		.@n = getarraysize(getvariableofnpc(getd(".d" + .@rk$ + "_quest"), .@npc$));
		for (.@i = 0; .@i < .@n; .@i++)
			if (getvariableofnpc(getd(".d" + .@rk$ + "_quest[" + .@i + "]"), .@npc$) == .@task)
				break;
		.@reward = getvariableofnpc(getd(".d" + .@rk$ + "_reward[" + .@i + "]"), .@npc$);
		if (checkweight(25051, .@reward) == 0) {
			mes "You're carrying too much to take the reward.";
			close;
		}
		erasequest .@q;
		setquest getvariableofnpc(.standby[.@r], .@npc$);
		getitem 25051, .@reward;
		mes "The receipt's in order. Here are your " + .@reward + " Union Insignias.";
		close;
	case 3:
		mes .@n$;
		mes "Which request do you want to give back? A bounty you give back stays closed until Wednesday's new posting.";
		next;
		switch (select("Wanted bounty:Delivery:Cancel")) {
		case 1:
			.@n = getarraysize(getvariableofnpc(.w_quest, .@npc$));
			for (.@i = 0; .@i < .@n; .@i++) {
				.@q = getvariableofnpc(.w_quest[.@i], .@npc$);
				if (isbegin_quest(.@q) == 1) {
					erasequest .@q;
					callfunc "F_Unity_WantedWait";
					mes .@n$;
					mes "Done. The bounty is back on the board.";
					close;
				}
			}
			mes .@n$;
			mes "You don't hold a bounty.";
			close;
		case 2:
			.@q = callfunc("F_Unity_HeldDelivery");
			mes .@n$;
			if (.@q == 0) {
				mes "You don't hold a delivery.";
				close;
			}
			erasequest .@q;
			mes "Done. You can take another delivery from the board.";
			close;
		}
		close;
	case 4:
		close;
	}
	end;

OnInit:
	if (callfunc("F_Unity_StandDown"))
		disablenpc strnpcinfo(0);
	end;
}

// The task quest for a held task or receipt quest. A delivery's receipt
// is its task quest + 1. Task quests are odd for A and C, even for B.
function	script	F_Unity_DeliveryTask	{
	.@q = getarg(0);
	if (.@q < 5577 || .@q > 5676)
		return (.@q % 2 == 1) ? .@q : .@q - 1;
	return (.@q % 2 == 0) ? .@q : .@q - 1;
}

// Starts the wait for the next bounty: quest 5470 until the next
// Wednesday 04:00 (GN, and the kRO client text of 5470).
function	script	F_Unity_WantedWait	{
	.@now = gettimetick(2);
	.@dow = gettime(DT_DAYOFWEEK);// 0 = Sunday
	.@sec = gettime(DT_HOUR) * 3600 + gettime(DT_MINUTE) * 60 + gettime(DT_SECOND);
	.@days = (WEDNESDAY - .@dow + 7) % 7;
	if (.@days == 0 && .@sec >= 4 * 3600)
		.@days = 7;
	unity_wanted_reset = .@now - .@sec + .@days * 86400 + 4 * 3600;
	setquest 5470;
	return;
}

// Delivery recipients. Name (iRO text), map and position (kRO and iRO
// quest text, matching bRO). [unverified: sprites; the kRO client's
// navigation table doesn't list these NPCs]
-	script	Delivery Recipient#cu	-1,{
	.@task = atoi(strnpcinfo(2));
	.@npc$ = "Unity Requests#cu";
	.@r = (.@task < 5577) ? 0 : ((.@task < 5676) ? 1 : 2);
	.@rk$ = getvariableofnpc(.rank$[.@r], .@npc$);
	.@n = getarraysize(getvariableofnpc(getd(".d" + .@rk$ + "_quest"), .@npc$));
	for (.@i = 0; .@i < .@n; .@i++)
		if (getvariableofnpc(getd(".d" + .@rk$ + "_quest[" + .@i + "]"), .@npc$) == .@task)
			break;
	.@item = getvariableofnpc(getd(".d" + .@rk$ + "_item[" + .@i + "]"), .@npc$);
	.@amt = getvariableofnpc(getd(".d" + .@rk$ + "_amount[" + .@i + "]"), .@npc$);
	.@n$ = "[" + strnpcinfo(1) + "]";
	mes .@n$;
	if (isbegin_quest(.@task) != 1) {
		mes "Hm? I'm waiting on a delivery from the City Merchant Union.";
		close;
	}
	if (countitem(.@item) < .@amt) {
		mes "You're from the union? I ordered " + .@amt + " " + getitemname(.@item) + ". Come back when you have them all.";
		close;
	}
	mes "My " + getitemname(.@item) + "! Thank you, I was starting to worry.";
	mes "Here's your receipt. Take it back to the union's clerk.";
	delitem .@item, .@amt;
	changequest .@task, .@task + 1;
	close;

OnInit:
	if (callfunc("F_Unity_StandDown"))
		disablenpc strnpcinfo(0);
	end;
}
